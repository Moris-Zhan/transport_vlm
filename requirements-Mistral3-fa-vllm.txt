# ============================================================
# Mistral3-fa-vllm conda 環境 — 完整套件與安裝/編譯步驟
# ============================================================
# 產生方式：/root/miniconda3/envs/Mistral3-fa-vllm/bin/pip freeze（2026-10-07，Python 3.10.22）
# 目標機器：NVIDIA H100 80GB ×2，driver 535（最高 CUDA 12.2），gcc 13.3
# 用途：同一個 env 同時能跑 LoRA 微調（flash_attention_2）與 vLLM 推論。
#
# 版本組合的關鍵限制（細節見 scripts/setup_flash_attn_env.sh、scripts/setup_vllm_env.sh 檔頭）：
#   - torch 用 cu126 build：driver 535 跑不動 cu130 build（requirements.txt 原本的 torch 2.13.0 是 cu130）。
#   - vllm 用 GitHub release 的 +cu129 build：PyPI 版是 CUDA 13 build。
#   - torchaudio / torchcodec 必須是 +cu126 build：PyPI 版連結 libcudart.so.13 / libnvrtc.so.13，
#     transformers import torchaudio 時就會讓 vllm serve 啟動失敗。
#   - flash-attn 官方預編譯 wheel 只到 torch 2.8，必須自己編譯。
#
# ------------------------------------------------------------
# 安裝步驟（在 repo root 執行）
# ------------------------------------------------------------
# 1. 建立 env：
#      /root/miniconda3/bin/conda create -y -n Mistral3-fa-vllm python=3.10
#      PY=/root/miniconda3/envs/Mistral3-fa-vllm/bin/python
#
# 2. 安裝 flash-attn 以外的全部套件（本檔已含 --extra-index-url，torch 系列會從 cu126 index 取得）：
#      $PY -m pip install -r requirements-Mistral3-fa-vllm.txt
#
# 3. 編譯 flash-attn 2.8.3.post1（只編 H100 的 sm90，108 核約 30 分鐘）：
#    3-1. 下載 NVIDIA CUDA 12.6 redistributable（nvcc + 標頭）到 env 底下當 CUDA_HOME，
#         不需要系統 CUDA toolkit，也不經過 conda（conda 的 cuda-nvcc 會強制裝 conda gcc 並跟 ld_impl 衝突）：
#           CUDA_DIR=/root/miniconda3/envs/Mistral3-fa-vllm/cuda-12.6; mkdir -p $CUDA_DIR
#           B=https://developer.download.nvidia.com/compute/cuda/redist
#           for p in cuda_nvcc/linux-x86_64/cuda_nvcc-linux-x86_64-12.6.85-archive \
#                    cuda_cudart/linux-x86_64/cuda_cudart-linux-x86_64-12.6.77-archive \
#                    cuda_cccl/linux-x86_64/cuda_cccl-linux-x86_64-12.6.77-archive \
#                    libcublas/linux-x86_64/libcublas-linux-x86_64-12.6.4.1-archive \
#                    libcusparse/linux-x86_64/libcusparse-linux-x86_64-12.5.4.2-archive \
#                    libcusolver/linux-x86_64/libcusolver-linux-x86_64-11.7.1.2-archive \
#                    libcurand/linux-x86_64/libcurand-linux-x86_64-10.3.7.77-archive \
#                    cuda_nvrtc/linux-x86_64/cuda_nvrtc-linux-x86_64-12.6.85-archive \
#                    cuda_profiler_api/linux-x86_64/cuda_profiler_api-linux-x86_64-12.6.77-archive \
#                    cuda_nvtx/linux-x86_64/cuda_nvtx-linux-x86_64-12.6.77-archive; do
#             curl -sSfL $B/$p.tar.xz | tar -xJ -C $CUDA_DIR --strip-components=1
#           done
#    3-2. 編譯安裝（--no-build-isolation：用 env 裡的 torch 編譯，ABI 才對得上；--no-deps：不動既有套件）：
#           CUDA_HOME=$CUDA_DIR PATH=$CUDA_DIR/bin:/root/miniconda3/envs/Mistral3-fa-vllm/bin:$PATH \
#           FLASH_ATTENTION_FORCE_BUILD=TRUE FLASH_ATTN_CUDA_ARCHS=90 MAX_JOBS=32 NVCC_THREADS=4 \
#             $PY -m pip install -v --no-build-isolation --no-deps flash-attn==2.8.3.post1
#
#    （也可以用腳本分層建置：Mistral3 → scripts/setup_flash_attn_env.sh → Mistral3-fa
#      → scripts/setup_vllm_env.sh → Mistral3-fa-vllm，兩支腳本都含驗證。）
#
# 4. 驗證：
#      $PY --version
#      $PY -c "import openai; print('openai:', openai.__version__)"
#      $PY -c "import torch; print('torch:', torch.__version__, torch.version.cuda, torch.cuda.is_available())"
#      $PY -c "import torch; print('torch_ABI:',torch._C._GLIBCXX_USE_CXX11_ABI)"
#      $PY -c "import vllm; print('vllm:',vllm.__version__)"
#      $PY -c "import transformers; print('transformers:', transformers.__version__)"
#      $PY -c "import xformers; print('xformers:', xformers.__version__)"
#      $PY -c "import flash_attn; print('flash_attn:', flash_attn.__version__)"
#    預期：3.10.22 / openai 3.10.0 / torch 2.13.0+cu126 12.6 True / True / vllm 0.28.0 /
#          transformers 5.16.1 / xformers 0.0.35 / flash_attn 2.8.3.post1
